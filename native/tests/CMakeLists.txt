cmake_minimum_required(VERSION 3.26)
project(cinderhell_audio_tests LANGUAGES C CXX)

# Standalone NDK build: no Gradle, SDL, APK, or physical audio output.
if(NOT ANDROID)
    message(FATAL_ERROR "Configure with the Android NDK toolchain")
endif()
set(LIBTYPE STATIC CACHE STRING "" FORCE)
set(ALSOFT_UTILS OFF CACHE BOOL "" FORCE)
set(ALSOFT_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ALSOFT_TESTS OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL OFF CACHE BOOL "" FORCE)
set(ALSOFT_BACKEND_OPENSL OFF CACHE BOOL "" FORCE)
set(ALSOFT_BACKEND_OBOE OFF CACHE BOOL "" FORCE)
set(ALSOFT_BACKEND_WAVE OFF CACHE BOOL "" FORCE)
set(ALSOFT_EMBED_HRTF_DATA OFF CACHE BOOL "" FORCE)
add_subdirectory(../../third_party/openal-soft openal-soft EXCLUDE_FROM_ALL)

add_executable(cinderhell_audio_test audio_loopback_test.c ../cinderhell_audio.c)
target_include_directories(cinderhell_audio_test PRIVATE ..)
target_compile_features(cinderhell_audio_test PRIVATE c_std_11)
target_compile_options(cinderhell_audio_test PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(cinderhell_audio_test PRIVATE AL_ALEXT_PROTOTYPES)
target_link_libraries(cinderhell_audio_test PRIVATE OpenAL::OpenAL m)
